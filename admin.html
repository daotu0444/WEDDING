Phản hồi từ khách mời đã tự động lưu trên đám mây Vercel. Nếu muốn lưu thêm bản sao sang Google Sheets, hãy dán URL Google Apps Script Web App bên dưới rồi bấm Lưu thay đổi.

📋 Hướng dẫn thiết lập Google Sheets miễn phí & Mã Google Apps Script (copy 1 chạm)
  1. Tạo một file Google Sheets mới trên Google Drive.
  2. Vào menu Tiện ích mở rộng (Extensions) → Apps Script, xóa mã mặc định và dán đoạn mã dưới đây vào.
  3. Bấm Triển khai (Deploy) → Tùy chọn triển khai mới (New deployment) → Ứng dụng web (Web app).
  4. Chọn Thực thi với tư cách: Tôi (Me) và Ai có quyền truy cập: Bất kỳ ai (Anyone) → Deploy → Copy URL Web App dán vào ô bên trên.
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheetName = data.type === 'rsvp' ? 'RSVP' : 'Wishes';
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  if (sheet.getLastRow() === 0) {
    if (sheetName === 'RSVP') {
      sheet.appendRow(['ID', 'Ten', 'TrangThai', 'SoKhach', 'LoiNhan', 'ThoiGian', 'MauThiep']);
    } else {
      sheet.appendRow(['ID', 'Ten', 'LoiChuc', 'ThoiGian', 'MauThiep']);
    }
  }
  if (sheetName === 'RSVP') {
    sheet.appendRow([data.id, data.name, data.status, data.guests, data.note, data.createdAt, data.templateId || '']);
  } else {
    sheet.appendRow([data.id, data.name, data.message, data.time, data.templateId || '']);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  function readSheet(name, mapFn) {
    var s = ss.getSheetByName(name);
    if (!s || s.getLastRow() < 2) return [];
    var rows = s.getDataRange().getValues().slice(1);
    return rows.map(mapFn).reverse();
  }
  var rsvps = readSheet('RSVP', function(r) {
    return { id: String(r[0]), name: String(r[1]), status: String(r[2]), guests: Number(r[3] || 0), note: String(r[4] || ''), createdAt: String(r[5] || ''), templateId: String(r[6] || '') };
  });
  var wishes = readSheet('Wishes', function(r) {
    return { id: String(r[0]), name: String(r[1]), message: String(r[2] || ''), time: String(r[3] || ''), templateId: String(r[4] || '') };
  });
  return ContentService.createTextOutput(JSON.stringify({ rsvps: rsvps, wishes: wishes }))
    .setMimeType(ContentService.MimeType.JSON);
}
